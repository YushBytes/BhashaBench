# Runtime dependencies for the BhashaBench backend.
#
# Pinned with compatible-release (~=) so patch releases are picked up while
# breaking minor releases are not. Versions below are the ones this phase was
# built and tested against.
#
# `uvicorn` is intentionally installed WITHOUT the `[standard]` extra: those
# extras (httptools, uvloop, watchfiles) ship no wheels for the newest CPython
# releases and fall back to compiling from source, which breaks a plain
# `pip install` on a developer machine running Python 3.13+. Plain uvicorn
# serves fine and `--reload` degrades to polling-based file watching. Add
# `uvicorn[standard]` deliberately when optimising a production deployment.

# -- Web framework ------------------------------------------------------------
fastapi~=0.162.0
uvicorn~=0.54.0
pydantic~=2.14.0
pydantic-settings~=2.16.0

# -- Persistence (Phase 2) ----------------------------------------------------
# SQLAlchemy 2.1 keeps the 2.0-style typed ORM API this project uses
# (`DeclarativeBase`, `Mapped`, `mapped_column`, `select()`).
sqlalchemy~=2.1.0
alembic~=1.20.0
# `psycopg` is psycopg 3. The `[binary]` extra ships prebuilt wheels so no
# local libpq or C toolchain is needed - important on Windows and in the slim
# container image. The driver is used synchronously; see docs/ARCHITECTURE.md
# for why this phase does not adopt an async engine.
psycopg[binary]~=3.3.0

# -- Scenario definitions (Phase 2) -------------------------------------------
# Loaded exclusively with `yaml.safe_load`; never `yaml.load`.
pyyaml~=6.0.0

# -- Speech-to-text providers (Phase 4) ---------------------------------------
# `httpx2` (not `httpx`) is the HTTP client this project standardises on: it is
# what Starlette 1.10+ requires for its TestClient, and running one client
# library rather than two keeps the transport behaviour in tests identical to
# production. Previously a dev-only dependency; the Prisma batch adapter now
# imports it at runtime.
httpx2~=2.16.0
# WebSocket *client*, for Prisma's streaming endpoint. Already present
# transitively, but declared explicitly: an adapter that depends on another
# package's transitive dependency breaks the day that package drops it.
websockets~=17.4
# Parses `multipart/form-data`, which is how audio is uploaded to
# `POST /api/v1/transcription` and how the Prisma REST contract accepts it.
# FastAPI raises at import time without it, so it is a hard runtime dependency
# rather than an optional extra.
python-multipart~=0.0.20

# -- Command-line interface (Phase 3) -----------------------------------------
# Typer builds the CLI from type hints, which keeps the command signatures and
# the validation in one place rather than duplicating argument parsing. It is
# a thin layer over Click, already an indirect dependency via uvicorn.
typer~=0.27.0
